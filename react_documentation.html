2. Structure d'un composant

Un composant React est une fonction qui retourne du JSX.

// ✅ Composant fonctionnel (standard actuel)
function MonComposant() {
  return (
    <div>
      <h1>Bonjour !</h1>
    </div>
  )
}

export default MonComposant

Nommage

// ✅ PascalCase obligatoire — sinon React le traite comme un tag HTML
function BoutonPrincipal() { ... }

// ❌ Mauvais — interprété comme <boutonprincipal> (tag inconnu)
function boutonprincipal() { ... }

Imports & Exports

// Export par défaut (un seul par fichier)
export default function App() { ... }

// Export nommé (plusieurs possibles)
export function Header() { ... }
export function Footer() { ... }

// Import
import App from './App'             // défaut
import { Header, Footer } from './Layout'  // nommés

3. JSX — Les règles

JSX est du JavaScript qui ressemble à du HTML.

function Exemple() {
  const nom = "React"
  const estConnecte = true
  const style = { color: 'blue', fontSize: '18px' }

  return (
    // ✅ Un seul élément racine (ou Fragment)
    <div>
      {/* Commentaire en JSX */}

      {/* Expression JavaScript entre accolades */}
      <h1>Bonjour {nom} !</h1>

      {/* className au lieu de class */}
      <p className="texte-principal">Paragraphe</p>

      {/* htmlFor au lieu de for */}
      <label htmlFor="email">Email</label>

      {/* Style en objet camelCase */}
      <p style={style}>Texte stylé</p>
      <p style={{ color: 'red', marginTop: '10px' }}>Inline direct</p>

      {/* Attributs booléens */}
      <input disabled />
      <input disabled={true} />

      {/* Self-closing obligatoire */}
      <input type="text" />
      <br />
      <img src="photo.jpg" alt="description" />
    </div>
  )
}

Fragment — éviter un div inutile

import { Fragment } from 'react'

function Exemple() {
  return (
    // Syntaxe courte <>
    <>
      <h1>Titre</h1>
      <p>Paragraphe</p>
    </>
  )
}

// Avec clé (listes) : utiliser Fragment explicite
function Liste({ items }) {
  return items.map(item => (
    <Fragment key={item.id}>
      <dt>{item.terme}</dt>
      <dd>{item.definition}</dd>
    </Fragment>
  ))
}

5. State & useState

Le state est une donnée locale au composant qui, quand elle change, re-rend le composant.

import { useState } from 'react'

function Compteur() {
  // [valeurActuelle, fonctionDeMiseAJour] = useState(valeurInitiale)
  const [count, setCount] = useState(0)
  const [nom, setNom] = useState('')
  const [actif, setActif] = useState(false)
  const [liste, setListe] = useState([])

  return (
    <div>
      <p>Compteur : {count}</p>
      <button onClick={() => setCount(count + 1)}>+1</button>
      <button onClick={() => setCount(prev => prev - 1)}>-1</button>
      {/* ✅ Utiliser la forme fonctionnelle quand la nouvelle valeur dépend de l'ancienne */}
    </div>
  )
}

State avec objet

function Formulaire() {
  const [user, setUser] = useState({ nom: '', email: '', age: 0 })

  function handleChange(e) {
    const { name, value } = e.target
    // ✅ Toujours créer un nouvel objet (ne pas muter)
    setUser(prev => ({ ...prev, [name]: value }))
  }

  return (
    <form>
      <input name="nom" value={user.nom} onChange={handleChange} />
      <input name="email" value={user.email} onChange={handleChange} />
    </form>
  )
}

State avec tableau

function ListeTodos() {
  const [todos, setTodos] = useState(['Apprendre React'])

  // Ajouter
  const ajouter = (item) => setTodos(prev => [...prev, item])

  // Supprimer

    setTodos(prev => prev.filter((_, i) => i !== index))

  // Modifier
  const modifier = (index, nouvelItem) =>
    setTodos(prev => prev.map((item, i) => i === index ? nouvelItem : item))

  return (
    <ul>
      {todos.map((todo, i) => (
        <li key={i}>
          {todo}
          <button onClick={() => supprimer(i)}>✕</button>
        </li>
      ))}
    </ul>
  )
}

6. useEffect

Exécute du code après le rendu — pour les effets de bord (fetch, abonnements, timers…).

import { useState, useEffect } from 'react'

function Exemple() {
  const [data, setData] = useState(null)
  const [userId, setUserId] = useState(1)

  // ✅ [] — s'exécute UNE SEULE FOIS (au montage)
  useEffect(() => {
    console.log('Composant monté')
  }, [])

  // ✅ [userId] — s'exécute quand userId change
  useEffect(() => {
    console.log('userId a changé :', userId)
  }, [userId])

  // ✅ Sans tableau — s'exécute à chaque rendu (rare)
  useEffect(() => {
    document.title = `User #${userId}`
  })

  // ✅ Avec nettoyage (cleanup) — important pour éviter les fuites mémoire
  useEffect(() => {
    const timer = setInterval(() => console.log('tick'), 1000)
    const handleResize = () => console.log('resize')
    window.addEventListener('resize', handleResize)

    // Fonction de nettoyage appelée au démontage ou avant le prochain effet
    return () => {
      clearInterval(timer)
      window.removeEventListener('resize', handleResize)
    }
  }, [])

  return <div>{data}</div>
}

useEffect avec fetch (pattern correct)

function Utilisateur({ id }) {
  const [user, setUser] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    let ignore = false  // ✅ Évite les race conditions

    async function fetchUser() {
      try {
        setLoading(true)
        const res = await fetch(`https://jsonplaceholder.typicode.com/users/${id}`)
        if (!res.ok) throw new Error('Erreur réseau')
        const data = await res.json()
        if (!ignore) setUser(data)  // N'update que si le composant est encore monté
      } catch (err) {
        if (!ignore) setError(err.message)
      } finally {
        if (!ignore) setLoading(false)
      }
    }

    fetchUser()

    return () => { ignore = true }  // Nettoyage
  }, [id])  // Re-fetch si id change

  if (loading) return <p>Chargement...</p>
  if (error) return <p>Erreur : {error}</p>
  return <p>{user?.name}</p>
}

7. useRef

Accéder à un élément DOM ou stocker une valeur sans déclencher de re-rendu.

import { useRef, useEffect } from 'react'

function ChampFocus() {
  const inputRef = useRef(null)

  useEffect(() => {
    inputRef.current.focus()  // Focus automatique au montage
  }, [])

  return <input ref={inputRef} placeholder="Je suis focusé !" />
}

useRef comme variable persistante

function Chronometre() {
  const [temps, setTemps] = useState(0)
  const intervalRef = useRef(null)  // Stocke l'id du timer sans re-rendre

  function demarrer() {
    intervalRef.current = setInterval(() => {
      setTemps(prev => prev + 1)
    }, 1000)
  }

  function arreter() {
    clearInterval(intervalRef.current)
  }

  return (
    <div>
      <p>{temps}s</p>
      <button onClick={demarrer}>Start</button>
      <button onClick={arreter}>Stop</button>
    </div>
  )
}

Stocker la valeur précédente

function usePrevious(valeur) {
  const ref = useRef()
  useEffect(() => { ref.current = valeur })
  return ref.current
}

function Composant({ count }) {
  const precedent = usePrevious(count)
  return <p>Avant : {precedent} — Maintenant : {count}</p>
}

8. useContext

Partager des données dans tout l'arbre de composants sans prop drilling.

import { createContext, useContext, useState } from 'react'

// 1. Créer le contexte
const ThemeContext = createContext(null)

// 2. Créer le Provider
function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('clair')

  const toggleTheme = () =>
    setTheme(prev => prev === 'clair' ? 'sombre' : 'clair')

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  )
}

// 3. Consommer dans n'importe quel composant enfant
function BoutonTheme() {
  const { theme, toggleTheme } = useContext(ThemeContext)

  return (
    <button onClick={toggleTheme}>
      Thème actuel : {theme}
    </button>
  )
}

// 4. Utilisation
function App() {
  return (
    <ThemeProvider>
      <BoutonTheme />
    </ThemeProvider>
  )
}

Custom hook pour le contexte (bonne pratique)

// ✅ Encapsuler useContext dans un hook custom
function useTheme() {
  const context = useContext(ThemeContext)
  if (!context) throw new Error('useTheme doit être utilisé dans ThemeProvider')
  return context
}

// Utilisation simplifiée
function MonComposant() {
  const { theme } = useTheme()
  return <div>{theme}</div>
}

9. useReducer

Alternative à useState pour les états complexes avec plusieurs actions.

import { useReducer } from 'react'

// 1. Définir le reducer (état, action) => nouvelÉtat
function reducer(state, action) {
  switch (action.type) {
    case 'INCREMENTER':
      return { ...state, count: state.count + 1 }
    case 'DECREMENTER':
      return { ...state, count: state.count - 1 }
    case 'RESET':
      return { ...state, count: 0 }
    case 'SET_NOM':
      return { ...state, nom: action.payload }
    default:
      throw new Error(`Action inconnue : ${action.type}`)
  }
}

// 2. État initial
const etatInitial = { count: 0, nom: '' }

// 3. Utiliser dans le composant
function Compteur() {
  const [state, dispatch] = useReducer(reducer, etatInitial)

  return (
    <div>
      <p>Compteur : {state.count}</p>
      <button onClick={() => dispatch({ type: 'INCREMENTER' })}>+</button>
      <button onClick={() => dispatch({ type: 'DECREMENTER' })}>-</button>
      <button onClick={() => dispatch({ type: 'RESET' })}>Reset</button>
      <input
        value={state.nom}
        onChange={e => dispatch({ type: 'SET_NOM', payload: e.target.value })}
      />
    </div>
  )
}

10. useMemo & useCallback

Optimisation : mémoriser des calculs ou des fonctions pour éviter des recalculs inutiles.

useMemo — mémoriser un calcul coûteux

import { useMemo, useState } from 'react'

function ListeFiltree({ items, recherche }) {
  // ✅ Ne recalcule que si items ou recherche change
  const itemsFiltres = useMemo(() => {
    console.log('Filtrage en cours...')
    return items.filter(item =>
      item.nom.toLowerCase().includes(recherche.toLowerCase())
    )
  }, [items, recherche])

  return (
    <ul>
      {itemsFiltres.map(item => <li key={item.id}>{item.nom}</li>)}
    </ul>
  )
}

useCallback — mémoriser une fonction

import { useCallback, useState } from 'react'

function Parent() {
  const [count, setCount] = useState(0)
  const [items, setItems] = useState([])

  // ✅ Référence stable — Enfant ne re-rend pas inutilement
  const ajouterItem = useCallback((item) => {
    setItems(prev => [...prev, item])
  }, [])  // Pas de dépendances car utilise la forme fonctionnelle

  return (
    <>
      <button onClick={() => setCount(c => c + 1)}>Parent re-rend ({count})</button>
      <Enfant onAjouter={ajouterItem} />
    </>
  )
}

// React.memo empêche le re-rendu si les props n'ont pas changé
const Enfant = React.memo(function Enfant({ onAjouter }) {
  console.log('Enfant rendu')
  return <button onClick={() => onAjouter('nouvel item')}>Ajouter</button>
})

12. Custom Hooks

Extraire de la logique réutilisable dans des fonctions préfixées use.

useLocalStorage

import { useState, useEffect } from 'react'

function useLocalStorage(cle, valeurInitiale) {
  const [valeur, setValeur] = useState(() => {
    try {
      const item = window.localStorage.getItem(cle)
      return item ? JSON.parse(item) : valeurInitiale
    } catch {
      return valeurInitiale
    }
  })

  useEffect(() => {
    window.localStorage.setItem(cle, JSON.stringify(valeur))
  }, [cle, valeur])

  return [valeur, setValeur]
}

// Utilisation
function App() {
  const [nom, setNom] = useLocalStorage('nom', '')
  return <input value={nom} onChange={e => setNom(e.target.value)} />
}

useFetch

import { useState, useEffect } from 'react'

function useFetch(url) {
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    let ignore = false
    setLoading(true)
    setError(null)

    fetch(url)
      .then(res => {
        if (!res.ok) throw new Error(`Erreur ${res.status}`)
        return res.json()
      })
      .then(data => { if (!ignore) setData(data) })
      .catch(err => { if (!ignore) setError(err.message) })
      .finally(() => { if (!ignore) setLoading(false) })

    return () => { ignore = true }
  }, [url])

  return { data, loading, error }
}

// Utilisation
function Posts() {
  const { data, loading, error } = useFetch('https://jsonplaceholder.typicode.com/posts')

  if (loading) return <p>Chargement...</p>
  if (error) return <p>Erreur : {error}</p>
  return <ul>{data?.map(p => <li key={p.id}>{p.title}</li>)}</ul>
}

useToggle

import { useState, useCallback } from 'react'

function useToggle(initial = false) {
  const [valeur, setValeur] = useState(initial)
  const toggle = useCallback(() => setValeur(v => !v), [])
  const setTrue = useCallback(() => setValeur(true), [])
  const setFalse = useCallback(() => setValeur(false), [])
  return [valeur, toggle, setTrue, setFalse]
}

function Modal() {
  const [ouvert, toggle, ouvrir, fermer] = useToggle(false)
  return (
    <>
      <button onClick={ouvrir}>Ouvrir</button>
      {ouvert && (
        <div>
          <p>Contenu de la modal</p>
          <button onClick={fermer}>Fermer</button>
        </div>
      )}
    </>
  )
}

14. Conditions & Rendu conditionnel

function Panneau({ user, score, items }) {
  const estConnecte = !!user

  return (
    <div>
      {/* ✅ && — affiche si condition vraie */}
      {estConnecte && <p>Bienvenue {user.nom} !</p>}

      {/* ✅ Ternaire — si/sinon */}
      {estConnecte ? <BoutonDeconnexion /> : <BoutonConnexion />}

      {/* ✅ || — valeur par défaut */}
      <p>{user?.bio || 'Aucune biographie'}</p>

      {/* ✅ Nullish coalescing ?? (null ou undefined seulement) */}
      <p>{score ?? 'Score non disponible'}</p>

      {/* ✅ Variable intermédiaire (pour la lisibilité) */}
      {(() => {
        if (score > 90) return <span>🏆 Excellent</span>
        if (score > 70) return <span>👍 Bien</span>
        return <span>📚 À améliorer</span>
      })()}

      {/* ✅ Liste vide */}
      {items.length === 0
        ? <p>Aucun élément</p>
        : <ul>{items.map(i => <li key={i.id}>{i.nom}</li>)}</ul>
      }
    </div>
  )
}

15. Événements

function GestionEvenements() {
  // ✅ Événement reçoit un SyntheticEvent (wrapper React autour du natif)
  function handleClick(e) {
    e.preventDefault()      // Empêcher comportement par défaut
    e.stopPropagation()     // Stopper la propagation
    console.log(e.target)   // L'élément cliqué
  }

  function handleChange(e) {
    console.log(e.target.value)   // Valeur de l'input
    console.log(e.target.checked) // Pour checkbox
  }

  function handleKeyDown(e) {
    if (e.key === 'Enter') console.log('Entrée pressée')
    if (e.ctrlKey && e.key === 's') console.log('Ctrl+S')
  }

  function handleSubmit(e) {
    e.preventDefault()
    const formData = new FormData(e.target)
    console.log(Object.fromEntries(formData))
  }

  // Passer des arguments à un handler
  function handleItemClick(id) {
    console.log('Item cliqué :', id)
  }

  return (
    <form onSubmit={handleSubmit}>
      <input onChange={handleChange} onKeyDown={handleKeyDown} />
      <button onClick={handleClick}>Clic</button>
      <button onClick={() => handleItemClick(42)}>Item 42</button>
    </form>
  )
}

16. Formulaires

Formulaire contrôlé (recommandé)

function FormulaireInscription() {
  const [form, setForm] = useState({
    nom: '',
    email: '',
    motDePasse: '',
    role: 'utilisateur',
    newsletter: false,
  })
  const [erreurs, setErreurs] = useState({})

  function valider() {
    const nouvellesErreurs = {}
    if (!form.nom.trim()) nouvellesErreurs.nom = 'Nom requis'
    if (!form.email.includes('@')) nouvellesErreurs.email = 'Email invalide'
    if (form.motDePasse.length < 6) nouvellesErreurs.motDePasse = 'Min. 6 caractères'
    return nouvellesErreurs
  }

  function handleChange(e) {
    const { name, value, type, checked } = e.target
    setForm(prev => ({
      ...prev,
      [name]: type === 'checkbox' ? checked : value
    }))
  }

  function handleSubmit(e) {
    e.preventDefault()
    const errs = valider()
    if (Object.keys(errs).length > 0) {
      setErreurs(errs)
      return
    }
    console.log('Soumis :', form)
  }

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <input
          name="nom"
          value={form.nom}
          onChange={handleChange}
          placeholder="Nom"
        />
        {erreurs.nom && <span className="erreur">{erreurs.nom}</span>}
      </div>

      <input
        name="email"
        type="email"
        value={form.email}
        onChange={handleChange}
      />

      <input
        name="motDePasse"
        type="password"
        value={form.motDePasse}
        onChange={handleChange}
      />

      <select name="role" value={form.role} onChange={handleChange}>
        <option value="utilisateur">Utilisateur</option>
        <option value="admin">Admin</option>
      </select>

      <label>
        <input
          name="newsletter"
          type="checkbox"
          checked={form.newsletter}
          onChange={handleChange}
        />
        S'abonner à la newsletter
      </label>

      <button type="submit">S'inscrire</button>
    </form>
  )
}

// src/services/api.js
const BASE_URL = 'https://jsonplaceholder.typicode.com'

async function requete(endpoint, options = {}) {
  const config = {
    headers: { 'Content-Type': 'application/json', ...options.headers },
    ...options,
  }
  if (config.body && typeof config.body === 'object') {
    config.body = JSON.stringify(config.body)
  }

  const res = await fetch(`${BASE_URL}${endpoint}`, config)
  if (!res.ok) {
    const erreur = await res.json().catch(() => ({}))
    throw new Error(erreur.message || `Erreur HTTP ${res.status}`)
  }
  return res.json()
}

export const api = {
  getUsers: () => requete('/users'),
  getUser: (id) => requete(`/users/${id}`),
  createUser: (data) => requete('/users', { method: 'POST', body: data }),
  updateUser: (id, data) => requete(`/users/${id}`, { method: 'PUT', body: data }),
  deleteUser: (id) => requete(`/users/${id}`, { method: 'DELETE' }),
}

  }

  return (

      <ul>
        {users.map(u => <li key={u.id}>{u.name}</li>)}

19. Gestion des erreurs

Error Boundary (erreurs de rendu)

import { Component } from 'react'

class ErrorBoundary extends Component {
  constructor(props) {
    super(props)
    this.state = { erreur: null }
  }

  static getDerivedStateFromError(error) {
    return { erreur: error }
  }

  componentDidCatch(error, info) {
    console.error('Erreur capturée :', error, info)
  }

  render() {
    if (this.state.erreur) {
      return this.props.fallback || <p>Quelque chose s'est mal passé.</p>
    }
    return this.props.children
  }
}

// Utilisation
<ErrorBoundary fallback={<p>Erreur dans ce composant</p>}>
  <ComposantRisque />
</ErrorBoundary>

function Accordion({ children }) {
  const [ouvert, setOuvert] = useState(null)

  const { ouvert, setOuvert } = useContext(AccordionContext)

      {estOuvert && <div>{children}</div>}

  )
}

22. Anti-patterns à éviter

// ❌ Muter le state directement
state.items.push(newItem)         // FAUX
setItems(prev => [...prev, newItem]) // CORRECT

// ❌ useState dans une condition ou boucle
if (condition) {
  const [val, setVal] = useState(0) // INTERDIT — ordre des hooks doit être constant
}

// ❌ Oublier les dépendances dans useEffect
useEffect(() => {
  fetchData(userId) // userId utilisé mais pas dans le tableau
}, [])  // ❌ → [userId]

// ❌ useEffect pour synchroniser deux states
const [a, setA] = useState(0)
const [double, setDouble] = useState(0)
useEffect(() => setDouble(a * 2), [a]) // ❌
// ✅ Calculer pendant le rendu
const double = a * 2

// ❌ Créer des composants dans des composants
function Parent() {
  function Enfant() { return <p>...</p> } // Re-créé à chaque rendu !
  return <Enfant />
}

// ❌ useEffect pour gérer des événements
useEffect(() => {
  // Pas besoin d'un effect pour ça — utiliser un handler
  if (clicked) { doSomething() }
}, [clicked])

// ❌ Index comme clé dans une liste dynamique
items.map((item, index) => <li key={index}>{item}</li>)  // Problèmes si réordonné
items.map(item => <li key={item.id}>{item.nom}</li>)  // ✅

// ❌ Trop de state — préférer dériver les valeurs
const [prenom, setPrenom] = useState('')
const [nomComplet, setNomComplet] = useState('') // ❌ dépend de prenom
const nomComplet = `${prenom} ${nom}` // ✅ calculé
